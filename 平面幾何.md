pub mod plane_geometry {
    use std::marker::PhantomData;
    use num_traits::*;
    use num_complex::Complex;
    use proconio::source::Readable;

    pub trait Point2<T: Num + Clone>: Num + Clone {
        fn new_point2(x: T, y: T) -> Self;
        fn x(&self) -> T;
        fn y(&self) -> T;

        fn dot(&self, other: &Self) -> T { self.x() * other.x() + self.y() * other.y() }
        fn cross(&self, other: &Self) -> T { self.x() * other.y() - self.y() * other.x() }
        fn norm2(&self) -> T { self.dot(self) }
        fn norm(&self) -> T where T: Float { self.norm2().sqrt() }

        fn area_2(polygon: &[Self]) -> T where Self: Sized, T: Signed {
            let mut sum = T::zero();
            for i in 0 .. polygon.len() {
                let p = &polygon[i];
                let q = &polygon[(i + 1) % polygon.len()];
                sum = sum + p.cross(q);
            }
            sum.abs()
        }

        fn area(polygon: &[Self]) -> T where Self: Sized, T: Signed + From<u8> {
            Self::area_2(polygon) / T::from(2u8)
        }

        fn centroid(polygon: &[Self]) -> Self where Self: Sized + Clone, T: Signed + From<u8> {
            assert!(!polygon.is_empty());
            if polygon.len() <= 3 {
                let mut sum = Self::new_point2(T::zero(), T::zero());
                for point in polygon {
                    sum = sum + point.clone();
                }
                let c = T::from(polygon.len() as u8);
                return Self::new_point2(sum.x() / c.clone(), sum.y() / c);
            }
            let area2 = Self::area_2(polygon);
            let mut weighted_sum = Self::new_point2(T::zero(), T::zero());
            for i in 1 .. polygon.len() - 1 {
                let triangle = [polygon[0].clone(), polygon[i].clone(), polygon[i + 1].clone()];
                let weight = Self::area_2(&triangle);
                let point = Self::centroid(&triangle);
                weighted_sum = weighted_sum + Self::new_point2(point.x() * weight.clone(), point.y() * weight);
            }
            Self::new_point2(weighted_sum.x() / area2.clone(), weighted_sum.y() / area2)
        }
    }
    impl<T: Num + Clone> Point2<T> for Complex<T> {
        fn new_point2(x: T, y: T) -> Self { Self::new(x, y) }
        fn x(&self) -> T { self.re.clone() }
        fn y(&self) -> T { self.im.clone() }
    }

    #[allow(non_camel_case_types)]
    pub struct point2<T> { _phantom: PhantomData<T> }
    impl<T: Readable> Readable for point2<T> {
        type Output = Complex<T::Output>;
        fn read<R: std::io::prelude::BufRead, S: proconio::source::Source<R>>(source: &mut S) -> Self::Output {
            Complex::new(T::read(source), T::read(source))
        }
    }
}
pub use plane_geometry::*;
